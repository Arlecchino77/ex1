--создание таблиц
CREATE TABLE Роль (
ID_роли INT PRIMARY KEY IDENTITY(1,1),
Название_роли NVARCHAR(50) NOT NULL UNIQUE
);
GO
CREATE TABLE ПунктыВыдачи (
ID_пункта INT PRIMARY KEY IDENTITY(1,1),
Адрес NVARCHAR(255) NOT NULL UNIQUE
);
GO
CREATE TABLE Категория (
ID_категории INT PRIMARY KEY IDENTITY(1,1),
Название NVARCHAR(100) NOT NULL UNIQUE
);
GO
CREATE TABLE Поставщик (
ID_поставщика INT PRIMARY KEY IDENTITY(1,1),
Название NVARCHAR(100) NOT NULL
);
GO
CREATE TABLE Производитель (
ID_производителя INT PRIMARY KEY IDENTITY(1,1),
Название NVARCHAR(100) NOT NULL
);
GO
CREATE TABLE Пользователь (
ID_пользователя INT PRIMARY KEY IDENTITY(1,1),
ФИО NVARCHAR(100) NOT NULL,
Логин NVARCHAR(100) NOT NULL UNIQUE,
Пароль NVARCHAR(100) NOT NULL,
ID_роли INT NOT NULL,
CONSTRAINT FK_Пользователь_Роль FOREIGN KEY (ID_роли) 
REFERENCES Роль(ID_роли)
);
GO
CREATE TABLE Товар (
ID_товара INT PRIMARY KEY IDENTITY(1,1),
Артикул NVARCHAR(20) NOT NULL UNIQUE,
Наименование NVARCHAR(200) NOT NULL,
Единица_измерения NVARCHAR(20) NOT NULL DEFAULT N'шт.',
Цена DECIMAL(10,2) NOT NULL CHECK (Цена >= 0),
ID_поставщика INT NOT NULL,
ID_производителя INT NOT NULL,
ID_категории INT NOT NULL,
Скидка DECIMAL(5,2) DEFAULT 0 CHECK (Скидка >= 0 AND Скидка <= 
100),
Количество INT NOT NULL DEFAULT 0 CHECK (Количество >= 0),
Описание NVARCHAR(MAX),
Фото NVARCHAR(100),
CONSTRAINT FK_Товар_Поставщик FOREIGN KEY (ID_поставщика) 
REFERENCES Поставщик(ID_поставщика),
CONSTRAINT FK_Товар_Производитель FOREIGN KEY 
(ID_производителя) REFERENCES Производитель(ID_производителя),
CONSTRAINT FK_Товар_Категория FOREIGN KEY (ID_категории) 
REFERENCES Категория(ID_категории)
);
GO
CREATE TABLE Заказ (
ID_заказа INT PRIMARY KEY IDENTITY(1,1),
Дата_заказа DATE NOT NULL,
Дата_доставки DATE NOT NULL,
ID_пункта INT NOT NULL,
ID_пользователя INT NOT NULL,
Код_получения NVARCHAR(20) NOT NULL UNIQUE,
Статус NVARCHAR(50) NOT NULL,
CONSTRAINT FK_Заказ_Пункт FOREIGN KEY (ID_пункта) 
REFERENCES ПунктыВыдачи(ID_пункта),
CONSTRAINT FK_Заказ_Пользователь FOREIGN KEY (ID_пользователя) 
REFERENCES Пользователь(ID_пользователя),
CONSTRAINT CHK_Даты CHECK (Дата_доставки >= Дата_заказа)
);
GO
CREATE TABLE ДеталиЗаказа (
ID_детали INT PRIMARY KEY IDENTITY(1,1),
ID_заказа INT NOT NULL,
ID_товара INT NOT NULL,
Количество INT NOT NULL CHECK (Количество > 0),
CONSTRAINT FK_ДеталиЗаказа_Заказ FOREIGN KEY (ID_заказа) 
REFERENCES Заказ(ID_заказа) ON DELETE CASCADE,
CONSTRAINT FK_ДеталиЗаказа_Товар FOREIGN KEY (ID_товара) 
REFERENCES Товар(ID_товара),
CONSTRAINT UQ_Заказ_Товар UNIQUE (ID_заказа, ID_товара)
);
GO
